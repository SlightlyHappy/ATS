# 🚀 BearSystems HR Tool - Premium Frontend Development Guide

## 🎯 Executive Summary & Value Proposition

**BearSystems HR Tool** is a simplified AI-powered career development platform that helps candidates optimize their careers and provides HR professionals with advanced candidate analysis. The frontend consists of three core experiences:

### 🌟 **Core Value Propositions**

#### 🎯 **For Individual Candidates (Self-Service):**
- **Free Resume Analysis** - Upload resume + email, get instant career insights
- **AI Career Coach** - 10 free chat messages for personalized career guidance
- **Learning Pathways** - Discover skills gaps and development recommendations
- **Premium Access** - ₹7000 for 3 days of unlimited AI coaching and advanced features

#### 💼 **For HR Professionals & Recruiters (Account Required):**
- **Advanced Candidate Analysis** - Deep insights into resume quality and fit
- **Bulk Processing** - Upload and analyze hundreds of resumes efficiently  
- **AI-Powered Search** - Natural language candidate search and matching
- **Contact Bear Systems** - Get enterprise account through direct consultation

## 🔗 API Endpoints Reference

### 🔐 Authentication & User Management

#### Public Endpoints (No Auth Required)
```javascript
// User Registration
POST /api/auth/register
Body: {
  email: string,
  first_name: string,
  last_name: string,
  password: string,
  job_title?: string,
  department?: string,
  phone?: string
}

// User Login
POST /api/auth/login  
Body: { email: string, password: string }
Response: { access_token: string, token_type: "bearer" }

// Health Check
GET /api/health
GET /api/health/detailed
GET /api/health/rag
```

#### Protected Endpoints (Requires Bearer Token)
```javascript
// Current User Profile
GET /api/users/me
PUT /api/users/me

// Token Refresh
POST /api/auth/refresh
```

### 📤 File Upload & Resume Processing

#### Single Resume Upload
```javascript
// Basic Upload (For Candidates)
POST /api/upload
Headers: { Authorization: "Bearer <token>" }
Form Data: { 
  file: File, 
  candidate_name?: string 
}

// Enhanced Upload with Analysis
POST /api/upload_single_auto_analyze
Form Data: { file: File, candidate_name?: string }
Response: { 
  resume_id: number,
  task_id: string,
  progress_url: string,
  estimated_analysis_time: string
}

// Enhanced Upload with Quality Scoring
POST /api/enhanced_upload
Form Data: { 
  file: File, 
  candidate_name?: string,
  auto_analyze?: boolean 
}
```

#### Bulk Upload (HR Professionals)
```javascript
// Bulk ZIP Upload
POST /api/bulk_upload
Form Data: { 
  file: File (ZIP), 
  description?: string 
}
Response: {
  batch_id: string,
  upload_url: string,
  expires_in: number
}

// Enhanced Bulk with Duplicate Detection
POST /api/enhanced_bulk_upload
Form Data: { 
  file: File (ZIP), 
  description?: string,
  check_duplicates?: boolean 
}
```

### 📊 Resume Analysis & Results

```javascript
// Get Resume Analysis
GET /api/analysis/{resume_id}

// Trigger Analysis
POST /api/analyze/{resume_id}

// Re-analyze Resume
POST /api/reanalyze/{resume_id}

// Get Resume Details
GET /api/resumes/{resume_id}

// Get Raw Resume Text
GET /api/resumes/{resume_id}/text

// List User Resumes
GET /api/resumes?skip=0&limit=100

```

```javascript
// Job Progress (Real-time)

// Detailed Job Status
GET /api/jobs/status/{job_id}
```

### 🔐 **Authentication & Security Implementation**
```typescript
// JWT Token Management with Refresh
class AuthService {
  private tokenRefreshTimer: NodeJS.Timeout | null = null;

  async login(credentials: LoginCredentials): Promise<AuthResult> {
    const response = await fetch('/api/auth/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(credentials)
    });

    const data = await response.json();

    if (response.ok) {
      // Store tokens securely
      this.setTokens(data.access_token, data.refresh_token);
      
      // Schedule automatic refresh
      this.scheduleTokenRefresh(data.expires_in);
      
      // Fetch user profile
      const user = await this.getCurrentUser();
      
      return { success: true, user, token: data.access_token };
    } else {
      return { success: false, error: data.detail };
    }
  }

  private setTokens(accessToken: string, refreshToken?: string): void {
    // Use httpOnly cookies in production for security
    localStorage.setItem('access_token', accessToken);
    if (refreshToken) {
      localStorage.setItem('refresh_token', refreshToken);
    }
  }

  private scheduleTokenRefresh(expiresIn: number): void {
    // Refresh 5 minutes before expiration
    const refreshTime = (expiresIn - 300) * 1000;
    
    this.tokenRefreshTimer = setTimeout(async () => {
      await this.refreshToken();
    }, refreshTime);
  }

  async refreshToken(): Promise<void> {
    const refreshToken = localStorage.getItem('refresh_token');
    
    if (!refreshToken) {
      this.logout();
      return;
    }

    try {
      const response = await fetch('/api/auth/refresh', {
        method: 'POST',
        headers: { 
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${refreshToken}`
        }
      });

      const data = await response.json();
      
      if (response.ok) {
        this.setTokens(data.access_token, data.refresh_token);
        this.scheduleTokenRefresh(data.expires_in);
      } else {
        this.logout();
      }
    } catch (error) {
      console.error('Token refresh failed:', error);
      this.logout();
    }
  }

  logout(): void {
    if (this.tokenRefreshTimer) {
      clearTimeout(this.tokenRefreshTimer);
    }
    
    localStorage.removeItem('access_token');
    localStorage.removeItem('refresh_token');
    
    // Redirect to login page
    window.location.href = '/login';
  }
}
```

```typescript
function useRealTimeUpdates(userId: string) {
  const [socket, setSocket] = useState<Socket | null>(null);
  const { updateJobStatus, addNotification } = useAppStore();

  useEffect(() => {

      });
    });

        action: {

        }
      });
    });

        persistent: data.severity === 'error'
      });
    });

    setSocket(newSocket);

    return () => {
      newSocket.close();
    };
  }, [userId]);

  return socket;

// Optimistic Updates for Better UX
function useOptimisticUpdates() {
  const { candidates, updateCandidate } = useAppStore();

  const updateCandidateOptimistically = async (
    candidateId: string, 
    updates: Partial<Candidate>
  ) => {
    // Update UI immediately
    const previousState = candidates.find(c => c.id === candidateId);
    updateCandidate(candidateId, updates);

    try {
      // Sync with server
      const response = await fetch(`/api/candidates/${candidateId}`, {
        method: 'PUT',
        headers: { 
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${localStorage.getItem('access_token')}`
        },
        body: JSON.stringify(updates)
      });

      if (!response.ok) {
        throw new Error('Update failed');
      }

      const updatedCandidate = await response.json();
      updateCandidate(candidateId, updatedCandidate);
    } catch (error) {
      // Revert on error
      if (previousState) {
        updateCandidate(candidateId, previousState);
      }
      
      throw error;
    }
  };

  return { updateCandidateOptimistically };
}
```

```typescript
// Create these service files
const ApiServices = {
  'services/api.ts': 'Base API configuration with interceptors',
  'services/auth.ts': 'Authentication and token management', 
  'services/upload.ts': 'File upload with progress tracking',
  'services/candidates.ts': 'Candidate search and management',
  'services/websocket.ts': 'Real-time updates and notifications'
};

// Example API service structure
class ApiService {
  private baseURL = 'https://hrtv6backend-production.up.railway.app';
  private token = localStorage.getItem('access_token');

  async request<T>(endpoint: string, options?: RequestInit): Promise<T> {
    const response = await fetch(`${this.baseURL}${endpoint}`, {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.token}`,
        ...options?.headers
      }
    });

    if (!response.ok) {
      throw new ApiError(response.status, await response.text());
    }

    return response.json();
  }
}
```
