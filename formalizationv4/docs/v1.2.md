```python
        for rule in self.scaling_rules['scale_up_conditions']:
            if rule['condition'](current_metrics):
                scale_up_score += rule['weight']
                max_increase = max(max_increase, rule['max_workers_increase'])
        
        # Calculate scale-down score
        scale_down_score = 0
        min_workers = 6  # Default minimum
        
        for rule in self.scaling_rules['scale_down_conditions']:
            if rule['condition'](current_metrics):

        if scale_up_score > 0.6 and worker_count < 8:  # Max 8 workers
            new_workers = min(worker_count + max_increase, 8)
            return {'action': 'scale_up', 'target_workers': new_workers, 'confidence': scale_up_score}

            new_workers = max(worker_count - 1, min_workers)
            return {'action': 'scale_down', 'target_workers': new_workers, 'confidence': scale_down_score}
```

```python
                'memory_limit': '1.5GB',
                'cpu_limit': 2,
                'status': 'starting'
            }
            
            # Wait for worker to be ready
            await self.wait_for_worker_ready(worker_id, timeout=30)
            self.active_workers[worker_id]['status'] = 'active'
            
            logger.info(f"Scaled up: Started {worker_type} worker {worker_id}")
    
    async def scale_down_workers(self, target_count, worker_type='analysis'):
        """Gracefully scale down workers"""
        current_workers = [w for w in self.active_workers.items() if w[1]['type'] == worker_type]
        current_count = len(current_workers)
        
        workers_to_remove = current_count - target_count
        
        if workers_to_remove <= 0:
            return
        
        # Select workers for removal (prioritize idle workers)
        idle_workers = await self.get_idle_workers(worker_type)
        busy_workers = [w for w in current_workers if w[0] not in idle_workers]
        
        # Remove idle workers first
        for worker_id, worker_info in idle_workers[:workers_to_remove]:
            await self.graceful_worker_shutdown(worker_id)
            workers_to_remove -= 1
            
        # If we still need to remove workers, wait for busy ones to finish
        if workers_to_remove > 0:
            for worker_id, worker_info in busy_workers[:workers_to_remove]:
                await self.wait_for_worker_idle(worker_id, timeout=300)  # 5 minutes
                await self.graceful_worker_shutdown(worker_id)
    
    async def graceful_worker_shutdown(self, worker_id):
        """Gracefully shutdown a worker"""
        if worker_id not in self.active_workers:
            return
        
        worker_info = self.active_workers[worker_id]
        worker_info['status'] = 'shutting_down'
        
        try:
            # Signal worker to finish current tasks
            await worker_info['worker'].graceful_shutdown()
            
            # Wait for completion with timeout
            await asyncio.wait_for(
                worker_info['worker'].wait_for_shutdown(),
                timeout=60
            )
            
        except asyncio.TimeoutError:
            # Force shutdown if graceful shutdown takes too long
            await worker_info['worker'].force_shutdown()
            
        finally:
            # Clean up resources
            await self.cleanup_worker_resources(worker_id)
            del self.active_workers[worker_id]
            
            logger.info(f"Scaled down: Shutdown worker {worker_id}")
```

```python
class DynamicPriorityAdjuster:
    def __init__(self):
        self.priority_factors = {
            'user_tier': {
                'admin': 1.5,
                'premium': 1.2,
                'regular': 1.0,
                'trial': 0.8
            },
            'request_urgency': {
                'critical': 2.0,
                'high': 1.5,
                'normal': 1.0,
                'low': 0.5
            },
            'wait_time_bonus': {
                'max_bonus': 0.5,           # Maximum bonus for waiting
                'bonus_threshold': 300,     # Start bonus after 5 minutes
                'bonus_rate': 0.001         # Bonus per second waiting
            },
            'resource_cost': {
                'low_memory': 1.2,         # Boost for low-memory tasks
                'high_memory': 0.8,        # Reduce for high-memory tasks
                'quick_analysis': 1.1,     # Boost for quick tasks
                'complex_analysis': 0.9    # Reduce for complex tasks
            }
        }
    
    async def calculate_dynamic_priority(self, queue_item):
        """Calculate dynamic priority based on multiple factors"""
        base_priority = queue_item.base_priority
        
        # User tier factor
        user_tier_factor = self.priority_factors['user_tier'].get(
            queue_item.user.tier, 1.0
        )
        
        # Request urgency factor
        urgency_factor = self.priority_factors['request_urgency'].get(
            queue_item.urgency, 1.0
        )
        
        # Wait time bonus
        wait_time = (datetime.utcnow() - queue_item.created_at).total_seconds()
        wait_bonus = min(
            self.priority_factors['wait_time_bonus']['max_bonus'],
            max(0, (wait_time - self.priority_factors['wait_time_bonus']['bonus_threshold']) * 
                self.priority_factors['wait_time_bonus']['bonus_rate'])
        )
        
        # Resource cost factor
        estimated_resources = await self.estimate_resource_requirements(queue_item)
        resource_factor = self.calculate_resource_factor(estimated_resources)
        
        # System load factor
        system_load = await self.get_current_system_load()
        load_factor = self.calculate_load_factor(system_load, queue_item)
        
        # Calculate final priority
        dynamic_priority = (
            base_priority * 
            user_tier_factor * 
            urgency_factor * 
            resource_factor * 
            load_factor + 
            wait_bonus
        )
        
        return {
            'priority': dynamic_priority,
            'factors': {
                'base': base_priority,
                'user_tier': user_tier_factor,
                'urgency': urgency_factor,
                'wait_bonus': wait_bonus,
                'resource': resource_factor,
                'load': load_factor
            }
        }
```

```python
class LoadAwareProcessor:
    def __init__(self):
        self.load_strategies = {
            'critical_load': {        # >95% CPU or >90% Memory
                'max_concurrent': 1,
                'allowed_queues': ['critical'],
                'batch_size': 1,
                'timeout_reduction': 0.5
            },
            'high_load': {           # >80% CPU or >80% Memory
                'max_concurrent': 2,
                'allowed_queues': ['critical', 'admin'],
                'batch_size': 1,
                'timeout_reduction': 0.7
            },
            'medium_load': {         # >60% CPU or >70% Memory
                'max_concurrent': 4,
                'allowed_queues': ['critical', 'admin', 'premium', 'normal'],
                'batch_size': 2,
                'timeout_reduction': 0.9
            },
            'normal_load': {         # <60% CPU and <70% Memory
                'max_concurrent': 6,
                'allowed_queues': ['critical', 'admin', 'premium', 'normal', 'background'],
                'batch_size': 3,
                'timeout_reduction': 1.0
            },
            'low_load': {           # <40% CPU and <50% Memory
                'max_concurrent': 8,
                'allowed_queues': ['critical', 'admin', 'premium', 'normal', 'background', 'batch'],
                'batch_size': 4,
                'timeout_reduction': 1.2
            }
        }
    
    async def determine_load_strategy(self, system_metrics):
        """Determine processing strategy based on system load"""
        cpu_usage = system_metrics['cpu']['percent']
        memory_usage = system_metrics['memory']['percent']
        
        if cpu_usage > 95 or memory_usage > 90:
            return 'critical_load'
        elif cpu_usage > 80 or memory_usage > 80:
            return 'high_load'
        elif cpu_usage > 60 or memory_usage > 70:
            return 'medium_load'
        elif cpu_usage > 40 or memory_usage > 50:
            return 'normal_load'
        else:
            return 'low_load'
    
    async def execute_load_aware_processing(self, strategy_name):
        """Execute processing based on load strategy"""
        strategy = self.load_strategies[strategy_name]
        
        # Get items from allowed queues
        processing_items = []
        
        for queue_name in strategy['allowed_queues']:
            queue = self.queues[queue_name]
            
            # Calculate items to take from this queue
            queue_weight = self.queue_weights[queue_name]
            items_from_queue = max(1, int(strategy['batch_size'] * queue_weight))
            
            items = await self.get_priority_items_from_queue(
                queue, limit=items_from_queue
            )
            processing_items.extend(items)
        
        # Sort by dynamic priority
        processing_items.sort(key=lambda x: x.dynamic_priority, reverse=True)
        
        # Limit to max concurrent
        processing_items = processing_items[:strategy['max_concurrent']]
        
        # Process items with appropriate timeouts
        timeout_multiplier = strategy['timeout_reduction']
        
        tasks = []
        for item in processing_items:
            adjusted_timeout = item.base_timeout * timeout_multiplier
            task = asyncio.create_task(
                self.process_item_with_timeout(item, adjusted_timeout)
            )
            tasks.append(task)
        
        if tasks:
            await asyncio.gather(*tasks, return_exceptions=True)
```
